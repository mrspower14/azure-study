#1. Enable WSL 2 
###########windows에서 설치할 경우 1.2.3.4.5 까지..

#2. Enable ‘Virtual Machine Platform’ 
Enable-WindowsOptionalFeature -Online -FeatureName VirtualMachinePlatform
